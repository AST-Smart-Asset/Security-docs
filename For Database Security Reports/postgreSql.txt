-- =========================================
-- SMART ASSET INVENTORY
-- ERD 01 - ASSET REGISTRY & PROCUREMENT
-- PostgreSQL
-- =========================================

CREATE EXTENSION IF NOT EXISTS pgcrypto;


-- =========================================
-- 1. ASSET CATEGORIES
-- =========================================

CREATE TABLE asset_categories (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    name VARCHAR(255) NOT NULL UNIQUE,
    useful_life_months INTEGER
);


-- =========================================
-- 2. ASSET MODELS
-- =========================================

CREATE TABLE asset_models (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),

    category_id UUID NOT NULL,

    manufacturer VARCHAR(255),
    model_name VARCHAR(255) NOT NULL,

    CONSTRAINT fk_asset_models_category
        FOREIGN KEY (category_id)
        REFERENCES asset_categories(id)
        ON DELETE RESTRICT
        ON UPDATE CASCADE
);


-- =========================================
-- 3. SUPPLIERS
-- =========================================

CREATE TABLE suppliers (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),

    name VARCHAR(255) NOT NULL,
    contact_email VARCHAR(255)
);


-- =========================================
-- 4. PURCHASE ORDERS
-- =========================================

CREATE TABLE purchase_orders (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),

    supplier_id UUID NOT NULL,

    po_number VARCHAR(100) NOT NULL UNIQUE,
    order_date DATE NOT NULL,
    total_cost NUMERIC(14,2),

    CONSTRAINT fk_purchase_orders_supplier
        FOREIGN KEY (supplier_id)
        REFERENCES suppliers(id)
        ON DELETE RESTRICT
        ON UPDATE CASCADE
);


-- =========================================
-- 5. PURCHASE ORDER LINE ITEMS
-- =========================================

CREATE TABLE purchase_order_line_items (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),

    purchase_order_id UUID NOT NULL,
    asset_model_id UUID NOT NULL,

    quantity INTEGER NOT NULL DEFAULT 1,
    unit_cost NUMERIC(14,2) NOT NULL,

    CONSTRAINT fk_line_items_purchase_order
        FOREIGN KEY (purchase_order_id)
        REFERENCES purchase_orders(id)
        ON DELETE CASCADE
        ON UPDATE CASCADE,

    CONSTRAINT fk_line_items_asset_model
        FOREIGN KEY (asset_model_id)
        REFERENCES asset_models(id)
        ON DELETE RESTRICT
        ON UPDATE CASCADE,

    CONSTRAINT chk_line_items_quantity
        CHECK (quantity > 0),

    CONSTRAINT chk_line_items_unit_cost
        CHECK (unit_cost >= 0)
);


-- =========================================
-- 6. ASSETS
-- =========================================

CREATE TABLE assets (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),

    asset_tag VARCHAR(100) NOT NULL UNIQUE,
    qr_code_value VARCHAR(255) UNIQUE,
    serial_number VARCHAR(255) UNIQUE,

    category_id UUID NOT NULL,
    asset_model_id UUID,

    purchase_order_line_item_id UUID,

    purchase_date DATE,
    status VARCHAR(50) NOT NULL DEFAULT 'active',

    CONSTRAINT fk_assets_category
        FOREIGN KEY (category_id)
        REFERENCES asset_categories(id)
        ON DELETE RESTRICT
        ON UPDATE CASCADE,

    CONSTRAINT fk_assets_model
        FOREIGN KEY (asset_model_id)
        REFERENCES asset_models(id)
        ON DELETE RESTRICT
        ON UPDATE CASCADE,

    CONSTRAINT fk_assets_purchase_line
        FOREIGN KEY (purchase_order_line_item_id)
        REFERENCES purchase_order_line_items(id)
        ON DELETE SET NULL
        ON UPDATE CASCADE
);


-- =========================================
-- 7. WARRANTIES
-- =========================================

CREATE TABLE warranties (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),

    supplier_id UUID,

    provider VARCHAR(255) NOT NULL,

    start_date DATE NOT NULL,
    expiry_date DATE NOT NULL,

    CONSTRAINT fk_warranties_supplier
        FOREIGN KEY (supplier_id)
        REFERENCES suppliers(id)
        ON DELETE SET NULL
        ON UPDATE CASCADE,

    CONSTRAINT chk_warranty_dates
        CHECK (expiry_date >= start_date)
);


-- =========================================
-- 8. DOCUMENT FILES
-- =========================================

CREATE TABLE document_files (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),

    storage_key VARCHAR(500) NOT NULL UNIQUE,
    content_type VARCHAR(100),
    scan_status VARCHAR(50) DEFAULT 'pending',

    uploaded_by UUID
);


-- =========================================
-- 9. WARRANTY DOCUMENTS
-- =========================================

CREATE TABLE warranty_documents (
    warranty_id UUID NOT NULL,
    document_id UUID NOT NULL,

    PRIMARY KEY (warranty_id, document_id),

    CONSTRAINT fk_warranty_documents_warranty
        FOREIGN KEY (warranty_id)
        REFERENCES warranties(id)
        ON DELETE CASCADE
        ON UPDATE CASCADE,

    CONSTRAINT fk_warranty_documents_document
        FOREIGN KEY (document_id)
        REFERENCES document_files(id)
        ON DELETE CASCADE
        ON UPDATE CASCADE
);


-- =========================================
-- 10. ASSET WARRANTIES
-- =========================================

CREATE TABLE asset_warranties (
    asset_id UUID NOT NULL,
    warranty_id UUID NOT NULL,

    PRIMARY KEY (asset_id, warranty_id),

    CONSTRAINT fk_asset_warranties_asset
        FOREIGN KEY (asset_id)
        REFERENCES assets(id)
        ON DELETE CASCADE
        ON UPDATE CASCADE,

    CONSTRAINT fk_asset_warranties_warranty
        FOREIGN KEY (warranty_id)
        REFERENCES warranties(id)
        ON DELETE CASCADE
        ON UPDATE CASCADE
);


-- =========================================
-- 11. ASSET DOCUMENTS
-- =========================================

CREATE TABLE asset_documents (
    asset_id UUID NOT NULL,
    document_id UUID NOT NULL,

    document_purpose VARCHAR(100),

    PRIMARY KEY (asset_id, document_id),

    CONSTRAINT fk_asset_documents_asset
        FOREIGN KEY (asset_id)
        REFERENCES assets(id)
        ON DELETE CASCADE
        ON UPDATE CASCADE,

    CONSTRAINT fk_asset_documents_document
        FOREIGN KEY (document_id)
        REFERENCES document_files(id)
        ON DELETE CASCADE
        ON UPDATE CASCADE
);


-- =========================================
-- 12. INVOICES
-- =========================================

CREATE TABLE invoices (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),

    purchase_order_id UUID NOT NULL,

    invoice_number VARCHAR(100) NOT NULL UNIQUE,
    invoice_date DATE NOT NULL,
    total_cost NUMERIC(14,2),

    CONSTRAINT fk_invoices_purchase_order
        FOREIGN KEY (purchase_order_id)
        REFERENCES purchase_orders(id)
        ON DELETE RESTRICT
        ON UPDATE CASCADE
);


-- =========================================
-- 13. INVOICE DOCUMENTS
-- =========================================

CREATE TABLE invoice_documents (
    invoice_id UUID NOT NULL,
    document_id UUID NOT NULL,

    PRIMARY KEY (invoice_id, document_id),

    CONSTRAINT fk_invoice_documents_invoice
        FOREIGN KEY (invoice_id)
        REFERENCES invoices(id)
        ON DELETE CASCADE
        ON UPDATE CASCADE,

    CONSTRAINT fk_invoice_documents_document
        FOREIGN KEY (document_id)
        REFERENCES document_files(id)
        ON DELETE CASCADE
        ON UPDATE CASCADE
);

-- =========================================
-- 1. ASSET ASSIGNMENTS
-- =========================================

CREATE TABLE asset_assignments (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),

    asset_id UUID NOT NULL,
    location_id UUID NOT NULL,
    custodian_id UUID,

    assignment_status VARCHAR(50) NOT NULL DEFAULT 'active',

    effective_from TIMESTAMP NOT NULL,
    effective_to TIMESTAMP,

    CONSTRAINT fk_asset_assignments_asset
        FOREIGN KEY (asset_id)
        REFERENCES assets(id)
        ON DELETE CASCADE
        ON UPDATE CASCADE,

    CONSTRAINT chk_assignment_dates
        CHECK (
            effective_to IS NULL
            OR effective_to >= effective_from
        )
);


-- =========================================
-- 2. ONLY ONE ACTIVE ASSIGNMENT PER ASSET
-- =========================================

CREATE UNIQUE INDEX uq_asset_assignments_active
    ON asset_assignments(asset_id)
    WHERE assignment_status = 'active';


-- =========================================
-- 3. CUSTODY TRANSFERS
-- =========================================

CREATE TABLE custody_transfers (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),

    asset_id UUID NOT NULL,

    from_assignment_id UUID,
    from_location_id UUID,

    to_location_id UUID NOT NULL,
    to_custodian_id UUID,

    status VARCHAR(50) NOT NULL DEFAULT 'pending',

    approved_at TIMESTAMP,
    approved_by UUID,

    completed_at TIMESTAMP,

    CONSTRAINT fk_custody_transfers_asset
        FOREIGN KEY (asset_id)
        REFERENCES assets(id)
        ON DELETE CASCADE
        ON UPDATE CASCADE,

    CONSTRAINT fk_custody_transfers_assignment
        FOREIGN KEY (from_assignment_id)
        REFERENCES asset_assignments(id)
        ON DELETE SET NULL
        ON UPDATE CASCADE,

    CONSTRAINT chk_custody_transfer_dates
        CHECK (
            completed_at IS NULL
            OR approved_at IS NULL
            OR completed_at >= approved_at
        )
);


-- =========================================
-- 4. ASSET STATUS HISTORY
-- =========================================

CREATE TABLE asset_status_history (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),

    asset_id UUID NOT NULL,

    previous_status VARCHAR(50),
    new_status VARCHAR(50) NOT NULL,

    reason TEXT,

    changed_at TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP,

    changed_by UUID,

    CONSTRAINT fk_asset_status_history_asset
        FOREIGN KEY (asset_id)
        REFERENCES assets(id)
        ON DELETE CASCADE
        ON UPDATE CASCADE
);


-- =========================================
-- 5. ASSET RISK EVALUATIONS
-- =========================================

CREATE TABLE asset_risk_evaluations (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),

    asset_id UUID NOT NULL,

    risk_score NUMERIC(6,2),
    risk_band VARCHAR(50),

    reasons TEXT,

    method_type VARCHAR(100),

    CONSTRAINT fk_asset_risk_evaluations_asset
        FOREIGN KEY (asset_id)
        REFERENCES assets(id)
        ON DELETE CASCADE
        ON UPDATE CASCADE,

    CONSTRAINT chk_risk_score
        CHECK (
            risk_score IS NULL
            OR risk_score >= 0
        )
);


-- =========================================
-- 6. ASSET RISK REVIEWS
-- =========================================

CREATE TABLE asset_risk_reviews (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),

    risk_evaluation_id UUID NOT NULL,

    reviewed_by UUID,

    reviewed_at TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP,

    CONSTRAINT fk_asset_risk_reviews_evaluation
        FOREIGN KEY (risk_evaluation_id)
        REFERENCES asset_risk_evaluations(id)
        ON DELETE CASCADE
        ON UPDATE CASCADE
);

-- =========================================
-- APPEND-ONLY AUDIT CONTROL
-- Prevent UPDATE / DELETE on audit_events
-- =========================================

CREATE OR REPLACE FUNCTION prevent_audit_event_changes()
RETURNS TRIGGER
LANGUAGE plpgsql
AS $$
BEGIN
    RAISE EXCEPTION
        'audit_events is append-only: UPDATE and DELETE are not allowed';
END;
$$;


CREATE TRIGGER trg_audit_events_append_only
BEFORE UPDATE OR DELETE
ON audit_events
FOR EACH ROW
EXECUTE FUNCTION prevent_audit_event_changes();
